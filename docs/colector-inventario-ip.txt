 CÓMO VERIFICAR QUE EL COLECTOR QUEDÓ BIEN (sólo lectura)
===========================================================================

-- 1. Una fila por adaptador, sin duplicados por MAC dentro del equipo
SELECT IdActivoFijoEQ, MAC, COUNT(*) AS Veces
FROM dbo.ActivoFijoEquiposRed
WHERE Vigente = 1 AND MAC IS NOT NULL
GROUP BY IdActivoFijoEQ, MAC
HAVING COUNT(*) > 1;
-- Esperado: 0 filas.

-- 2. Un solo EsPrincipal por equipo
SELECT IdActivoFijoEQ, COUNT(*) AS Principales
FROM dbo.ActivoFijoEquiposRed
WHERE Vigente = 1 AND EsPrincipal = 1
GROUP BY IdActivoFijoEQ
HAVING COUNT(*) > 1;
-- Esperado: 0 filas (hoy: 0).

-- 3. Ningún dato técnico vacío
SELECT SUM(CASE WHEN IP IS NULL THEN 1 ELSE 0 END)          AS SinIP,
       SUM(CASE WHEN MAC IS NULL THEN 1 ELSE 0 END)         AS SinMAC,
       SUM(CASE WHEN Hostname IS NULL THEN 1 ELSE 0 END)    AS SinHostname,
       SUM(CASE WHEN TipoMedio IS NULL THEN 1 ELSE 0 END)   AS SinTipoMedio,
       SUM(CASE WHEN Mascara IS NULL THEN 1 ELSE 0 END)     AS SinMascara,
       SUM(CASE WHEN PuertaEnlace IS NULL THEN 1 ELSE 0 END) AS SinGateway
FROM dbo.ActivoFijoEquiposRed
WHERE Vigente = 1;
-- Hoy: 0 / 0 / 0 / 0 / 0 / 2 (los 2 sin gateway son WiFi sin ruta).

-- 4. Cobertura: activos que el colector todavía no ve
SELECT COUNT(*) AS ActivosSinRed
FROM dbo.ActivoFijoEquipos a
WHERE NOT EXISTS (SELECT 1 FROM dbo.ActivoFijoEquiposRed r
                   WHERE r.IdActivoFijoEQ = a.IdActivoFijoEQ
                     AND r.Vigente = 1);
-- Hoy: 35. No es un bug de formato: esos equipos no reportaron.

-- 5. Que la maestra quedó una sola fila por IP (esto lo hace la base)
SELECT COUNT(*) AS Filas, COUNT(DISTINCT IP) AS IPs
FROM dbo.InventarioIPMaestro;
-- Esperado: Filas = IPs. Hoy: 1623 y 1623.

-- 6. Que el colector NO duplicó IP en el inventario manual
SELECT COUNT(*) AS Duplicadas
FROM (SELECT IP FROM dbo.InventarioGeneralIP
      GROUP BY IP HAVING COUNT(*) > 1) d;
-- Esperado: 0.
